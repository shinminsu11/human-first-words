from kivy.app import App from kivy.uix.boxlayout import BoxLayout from kivy.uix.textinput import TextInput from kivy.uix.button import Button from kivy.uix.gridlayout import GridLayout class KeyboardApp(App): def build(self): # 전체 화면 레이아웃 (세로 배치) root_layout = BoxLayout(orientation='vertical', padding=20, spacing=20) # 텍스트 입력창 self.output = TextInput( text='', font_size=32, size_hint_y=None, height=100, multiline=False ) root_layout.add_widget(self.output) # 자판 버튼이 들어갈 그리드 레이아웃 (4열) key_layout = GridLayout(cols=4, spacing=10, size_hint_y=None, height=400) # 자판 글자 목록 keys = ['Q', 'W', 'E', 'R', 'T', 'Y', 'U', 'I', 'O', 'P', 'A', 'S', 'D', 'F', 'G', 'H', 'J', 'K', 'L', 'Z', 'X', 'C', 'V', 'B', 'N', 'M'] for char in keys: btn = Button( text=char, font_size=30, background_color=(0.9, 0.9, 0.9, 1) ) # 버튼 클릭 시 입력창에 글자 추가하는 함수 연결 btn.bind(on_press=lambda instance, c=char: self.type_key(c)) key_layout.add_widget(btn) # 지우기 버튼 del_btn = Button(text='DEL', font_size=24, background_color=(0.8, 0.4, 0.4, 1)) del_btn.bind(on_press=self.backspace_key) key_layout.add_widget(del_btn) # 공백 버튼 space_btn = Button(text='SPACE', font_size=24, background_color=(0.4, 0.6, 0.8, 1)) space_btn.bind(on_press=lambda instance: self.type_key(' ')) key_layout.add_widget(space_btn) root_layout.add_widget(key_layout) return root_layout def type_key(self, char): self.output.text += char def backspace_key(self, instance): self.output.text = self.output.text[:-1] if __name__ == '__main__': KeyboardApp().run()
